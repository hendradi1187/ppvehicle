#   python -m pip install paddlepaddle-gpu==3.3.1 -i https://www.paddlepaddle.org.cn/packages/stable/cu126/
# then:
#   python -m pip install -r requirements.txt --no-deps-for-paddle
#
# Practically: install requirements.txt first, then replace paddlepaddle:
#   pip uninstall -y paddlepaddle
#   pip install paddlepaddle-gpu==3.3.1 -i https://www.paddlepaddle.org.cn/packages/stable/cu126/
#
